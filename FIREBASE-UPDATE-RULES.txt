IMPORTANT

- These examples allow any signed-in EthioLegacy account to read active statuses and public channels. Statuses expire from the app's list after 24 hours; this client-side expiry does not automatically erase old Firestore documents. Configure Firestore TTL on expiresAt if you want database cleanup.

FIRESTORE RULES — merge these match blocks inside service cloud.firestore { match /databases/{database}/documents { ... } }

match /statuses/{statusId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.expiresAt is number;
  allow update: if request.auth != null && (
    resource.data.uid == request.auth.uid ||
    (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['viewers'])
      && request.resource.data.viewers is list
      && request.resource.data.viewers.hasAll(resource.data.viewers)
      && request.resource.data.viewers.hasAny([request.auth.uid]))
  );
  allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
}

match /channels/{channelId} {
  allow read: if request.auth != null && resource.data.isPublic == true;
  allow create: if request.auth != null
    && request.resource.data.ownerUid == request.auth.uid;
  allow update: if request.auth != null && (
    resource.data.ownerUid == request.auth.uid ||
    (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['followers'])
      && request.resource.data.followers is list)
  );
  allow delete: if request.auth != null && resource.data.ownerUid == request.auth.uid;

  match /posts/{postId} {
    function channelDoc() {
      return get(/databases/$(database)/documents/channels/$(channelId));
    }
    allow read: if request.auth != null
      && channelDoc().data.isPublic == true;
    allow create: if request.auth != null
      && channelDoc().data.ownerUid == request.auth.uid
      && request.resource.data.uid == request.auth.uid;
    allow update, delete: if request.auth != null
      && channelDoc().data.ownerUid == request.auth.uid;
  }
}

STORAGE RULES — merge inside service firebase.storage { match /b/{bucket}/o { ... } }

match /statuses/{userId}/{fileName} {
  allow read: if request.auth != null;
  allow write: if request.auth != null && request.auth.uid == userId
    && request.resource.size < 50 * 1024 * 1024;
}
match /channelPosts/{channelId}/{userId}/{fileName} {
  allow read: if request.auth != null;
  allow write: if request.auth != null && request.auth.uid == userId
    && request.resource.size < 50 * 1024 * 1024;
}

INVITATION LINKING + INCOMING CALLS — FIRESTORE ADD-ON
Merge these blocks INSIDE your existing service cloud.firestore / match /databases/{database}/documents block. Do not paste them into Firebase Storage Rules and do not replace your other rules.

// A joining user may create/update only their own contact entry (document ID == their UID)
// under the inviter's friends list. The inviter can read/manage their own friends list.
match /users/{ownerUid}/friends/{friendUid} {
  allow read, delete: if request.auth != null && request.auth.uid == ownerUid;
  allow create, update: if request.auth != null && (
    request.auth.uid == ownerUid ||
    (request.auth.uid == friendUid && request.resource.data.uid == request.auth.uid)
  );
}

// The inviter can read their own saved category. The invited user can write
// only their own membership document into the inviter's category.
match /users/{ownerUid}/specialGroups/{groupName}/members/{memberUid} {
  allow read, delete: if request.auth != null && request.auth.uid == ownerUid;
  allow create, update: if request.auth != null && (
    request.auth.uid == ownerUid ||
    (request.auth.uid == memberUid && groupName in ['love', 'family', 'friend', 'classmate'])
  );
}

// Incoming call listener queries calls by receiverId only; the app filters status locally.
match /calls/{callId} {
  allow create: if request.auth != null
    && request.resource.data.callerId == request.auth.uid
    && request.resource.data.receiverId is string;
  allow read: if request.auth != null && (
    resource.data.callerId == request.auth.uid ||
    resource.data.receiverId == request.auth.uid
  );
  allow update: if request.auth != null && (
    resource.data.callerId == request.auth.uid ||
    resource.data.receiverId == request.auth.uid
  );
  allow delete: if request.auth != null && resource.data.callerId == request.auth.uid;

  match /offerCandidates/{candidateId} {
    function parentCall() {
      return get(/databases/$(database)/documents/calls/$(callId));
    }
    allow read, write: if request.auth != null && parentCall().exists()
      && (parentCall().data.callerId == request.auth.uid || parentCall().data.receiverId == request.auth.uid);
  }
  match /answerCandidates/{candidateId} {
    function parentCall() {
      return get(/databases/$(database)/documents/calls/$(callId));
    }
    allow read, write: if request.auth != null && parentCall().exists()
      && (parentCall().data.callerId == request.auth.uid || parentCall().data.receiverId == request.auth.uid);
  }
}

IMPORTANT SECURITY NOTE
These rules allow a signed-in user to create only their own UID entry under another user's friends/specialGroups collection. They do not allow arbitrary edits to the other user's profile. Review these additions against your existing rules before publishing; overlapping match rules are OR-combined by Firestore.
